function updateStatus(index){ let orders = JSON.parse( localStorage.getItem("orders") ) || []; let statusList = [ "Pending", "Preparing", "Delivered" ]; let current = orders[index].status; let next = statusList[ (statusList.indexOf(current)+1) % statusList.length ]; orders[index].status = next; localStorage.setItem( "orders", JSON.stringify(orders) ); alert( "Status Updated: "+next ); showOrders(); }
